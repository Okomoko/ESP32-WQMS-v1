create table public.sensor_config (
  id integer not null,
  system text not null,
  sensor_name text not null,
  unit text not null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint sensor_config_pkey primary key (id)
) TABLESPACE pg_default;

create unique INDEX IF not exists idx_sensor_config_system_id on public.sensor_config using btree (system, id) TABLESPACE pg_default;

create table public.sensor_readings (
  id bigint generated by default as identity not null,
  system text not null,
  timestamp timestamp with time zone not null,
  sensor0 double precision null,
  sensor1 double precision null,
  sensor2 double precision null,
  sensor3 double precision null,
  sensor4 double precision null,
  sensor5 double precision null,
  sensor6 double precision null,
  sensor7 double precision null,
  created_at timestamp with time zone null default now(),
  constraint sensor_readings_pkey primary key (id)
) TABLESPACE pg_default;

create index IF not exists idx_readings_timestamp on public.sensor_readings using btree ("timestamp" desc) TABLESPACE pg_default;

create index IF not exists idx_readings_system on public.sensor_readings using btree (system) TABLESPACE pg_default;

create table public.system_logs (
  id bigint generated by default as identity not null,
  system text not null,
  timestamp timestamp with time zone not null,
  module text not null,
  level text not null,
  message text not null,
  created_at timestamp with time zone null default now(),
  constraint system_logs_pkey primary key (id)
) TABLESPACE pg_default;

create index IF not exists idx_logs_timestamp on public.system_logs using btree ("timestamp" desc) TABLESPACE pg_default;

create index IF not exists idx_logs_system on public.system_logs using btree (system) TABLESPACE pg_default;

create index IF not exists idx_logs_level on public.system_logs using btree (level) TABLESPACE pg_default;

create index IF not exists idx_logs_module on public.system_logs using btree (module) TABLESPACE pg_default;

create index IF not exists idx_logs_system_timestamp on public.system_logs using btree (system, "timestamp" desc) TABLESPACE pg_default;

create table public.system_config (
  id bigint generated by default as identity not null,
  system text not null,
  location text null,
  timezone text null,
  sample_interval_ms integer null,
  upload_interval_sec integer null,
  last_sensor_upload_ts bigint null,
  last_log_upload_ts bigint null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint system_config_pkey primary key (id),
  constraint system_config_system_key unique (system)
) TABLESPACE pg_default;
